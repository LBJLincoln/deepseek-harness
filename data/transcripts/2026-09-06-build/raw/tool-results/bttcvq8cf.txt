=== overlays/route-only.cordis.yml
# The bench for a plan whose arms all run on the session's own route: the same
# cells, claiming the isolation this host's executors actually enforce. The base
# composition claims `none` because its E2 candidate arm delegates to the
# product, and the runner refuses an out-of-process implementer under any higher
# claim; an arm set without one can claim what it enforces.
#
# Three patches make the claim provable rather than only declared. The runner
# states `process`; the barrier raises `isolationClaim` to the same level, which
# is what turns the composed subagent seam's refusal-to-start into
# `denied-at-executor` in the census and refuses every child start in an
# implementer session; and the search tool joins the composition because its
# confined ripgrep spawn is what registers `subprocess` enforcement — without it
# that capability is composed and unenforced, and a certificate over this
# composition is refused for it.
#
# Measured on 2026-09-08: with the Claude Code LLM route composed, every cell
# still fails its certificate with `VERIFICATION_ISOLATION_UNPROVEN` ("capability
# "subprocess" is composed without read-barrier enforcement"), because the route
# itself spawns the product CLI through the subprocess service outside the
# sandbox, and the census attributes that spawn to the implementer session. Until
# the census can tell the route's own spawn from a tool the implementer holds,
# route arms run on the base composition: the same bwrap seal enforced, the
# claim `none`. This overlay serves a route that spawns nothing.
- id: base
  name: '@deepseek-ai/cordis-plugin-include'
  config:
    path: ../cordis.yml
    patches:
      - id: environment-runner
        name: '@deepseek-ai/dsh-environment-runner'
        config:
          isolation: process
          maxAttempts: 3
          evidenceMaxChars: 2048
      - id: read-barrier
        name: '@deepseek-ai/dsh-read-barrier'
        config:
          isolationClaim: process
      # The base composes no search tool, so this one is inserted: a non-insert
      # patch only overrides an entry the base already names.
      - insert:
          - id: tool-fs-search
            name: '@deepseek-ai/dsh-tool-fs-search'
            config:
              sampleOverCapGlobResults: false
      # The village rule reads this file on its own: a composition that names the
      # runner must also name the session budget, the persistence backend, and
      # the checkpoint policy, so the base's three are restated unchanged.
      - id: budget-policy
        name: '@deepseek-ai/dsh-budget-policy'

          maxTotalTokens: 1500000
          maxWallMs: 1200000
      - id: persistence
        name: '@deepseek-ai/dsh-session-persistence-jsonl'

# in nothing else, which is what makes a pair over one route a measurement of
# the composition. The roots are found from the repository root above
# `baseUrl`, as the craft-skills overlay does.

  name: '@deepseek-ai/cordis-plugin-include'
  config:
    path: ../cordis.yml

# `includeUserRoot: false` keeps the machine's own `<dshHome>/.agent-presets`
# out of the roster: a bench arm names a preset by id, and a home directory that
# happened to supply that id would silently decide what the arm measured.
- id: agent-presets
  name: '@deepseek-ai/dsh-agent-presets'
  config:
    default: bench
    includeUserRoot: false
    roots:
      - trust: system
        path: !!js |
            (() => {
              const { fileURLToPath } = process.getBuiltinModule('node:url')
              const { existsSync } = process.getBuiltinModule('node:fs')
              const { dirname, join } = process.getBuiltinModule('node:path')
              let dir = fileURLToPath(baseUrl)
              while (!existsSync(join(dir, 'pnpm-workspace.yaml'))) {
                const parent = dirname(dir)
                if (parent === dir) throw new Error('proving-ground-bench overlay: repository root not found above ' + baseUrl)
                dir = parent
              }
              return join(dir, 'examples', 'headless-agent', 'tests', 'fixtures', 'proving-ground-bench', 'presets')
            })()
