    80	/** Serve one deterministic DeepSeek-compatible response while retaining its request body. */
    81	async function deepseekDefaultsServer(): Promise<DeepSeekDefaultsServer> {
    82	  const requests: JsonObject[] = []
    83	  const server = createServer((request: IncomingMessage, response: ServerResponse) => {
    84	    let body = ''
    85	    request.setEncoding('utf8')
    86	    request.on('data', (chunk: string) => { body += chunk })
    87	    request.on('end', () => {
    88	      requests.push(JSON.parse(body) as JsonObject)
    89	      response.writeHead(200, { 'content-type': 'text/event-stream' })
    90	      let keepAlives = 3
    91	      const write = (): void => {
    92	        if (keepAlives-- > 0) {
    93	          response.write(': keep-alive\n\n')
    94	          setTimeout(write, 60)
    95	          return
    96	        }
    97	        response.end([
    98	          'data: {"choices":[{"delta":{"content":"DEFAULTS_OK"}}]}',
    99	          'data: {"choices":[{"delta":{},"finish_reason":"stop"}],"usage":{"prompt_tokens":3,"completion_tokens":1}}',
   100	          'data: [DONE]',
   101	          '',
   102	        ].join('\n\n'))
   103	      }
   104	      setTimeout(write, 60)
   105	    })
   106	  })
   107	  await new Promise<void>(resolve => server.listen(0, '127.0.0.1', resolve))
   108	  const address = server.address()
   109	  if (address === null || typeof address === 'string') throw new Error('DeepSeek defaults snapshot server has no port')
   110	  return {
   111	    url: `http://127.0.0.1:${address.port}`,
   112	    requests,
   113	    close: () => new Promise(resolve => server.close(() => { resolve() })),
   114	  }
   115	}

   123	function contextFromLogs(contents: readonly string[]): NormalizeContext {
   124	  const headers = contents.map(content => parseJsonl(content)[0])
   125	  return {
   126	    sessionIds: headers.flatMap(header => typeof header?.id === 'string' ? [header.id] : []),
   127	    cwd: typeof headers[0]?.cwd === 'string' ? headers[0].cwd : '\0no-cwd\0',
   128	  }

   131	function normalizeHeadlessStream(rawStdout: string, cwd: string): string {
   132	  const records = parseJsonl(rawStdout)
   133	  if (records.length === 0) throw new Error('headless snapshot emitted no stream-json records')
   134	  const final = records.at(-1)
   135	  if (final?.type !== 'result') throw new Error('headless snapshot did not end with a result record')
   136	  if (records.slice(0, -1).some(record => record.type !== 'session_event')) {
   137	    throw new Error('headless snapshot emitted a non-event record before its result')
   138	  }
   139	
   140	  const sessionIds = [...new Set(records.flatMap(record => typeof record.sessionId === 'string' ? [record.sessionId] : []))]
   141	  if (sessionIds.length !== 1) throw new Error(`headless snapshot streamed ${sessionIds.length} main session ids`)
   142	  const context: NormalizeContext = { sessionIds, cwd }
   143	  const events = records.slice(0, -1).map((record) => {
   144	    if (record.event === null || typeof record.event !== 'object' || Array.isArray(record.event)) {
   145	      throw new Error('headless snapshot emitted an invalid session event')
   146	    }
   147	    return record.event as JsonObject
   148	  })
   149	  const normalizedEvents = parseJsonl(scrubRequestHeaders(normalizeSessionLog(
   150	    `${events.map(event => JSON.stringify(event)).join('\n')}\n`,
   151	    context,
   152	  )))
   153	  const normalizedRecords = records.map((record, index) => index < normalizedEvents.length
   154	    ? { ...record, event: normalizedEvents[index] }
   155	    : record)
   156	  return normalizeStdout(`${normalizedRecords.map(record => JSON.stringify(record)).join('\n')}\n`, context)
   157	}

   159	/** Zero durable goal timestamps inside both metadata records and rendered XML JSON. */
   160	function normalizeGoalTimestamps(value: unknown): unknown {
   161	  if (typeof value === 'string') {
   162	    return value.replace(/("(?:createdAt|updatedAt|clearedAt)":)\d+/g, '$10')
   163	  }
   164	  if (Array.isArray(value)) return value.map(normalizeGoalTimestamps)
   165	  if (value !== null && typeof value === 'object') {
   166	    return Object.fromEntries(Object.entries(value).map(([key, item]) => [
   167	      key,
   168	      ['createdAt', 'updatedAt', 'clearedAt'].includes(key) && typeof item === 'number'
   169	        ? 0
   170	        : normalizeGoalTimestamps(item),
   171	    ]))
   172	  }
   173	  return value
   174	}

   176	/** Normalize the stream's durable goal timestamps after the shared scrubbers. */
   177	function normalizeGoalStream(rawStdout: string, cwd: string): string {
   178	  return parseJsonl(normalizeHeadlessStream(rawStdout, cwd))
   179	    .map(record => JSON.stringify(normalizeGoalTimestamps(record)))
   180	    .join('\n') + '\n'
   181	}

   183	async function scenarioPrompt(dir: string, label: string): Promise<string> {
   184	  const input = JSON.parse(await readFile(join(dir, 'input.json'), 'utf8')) as {
   185	    steps?: { op?: unknown; text?: unknown }[]
   186	  }
   187	  const prompt = input.steps?.find(step => step.op === 'prompt')?.text
   188	  if (typeof prompt !== 'string') throw new Error(`${label} input has no prompt step`)
   189	  return prompt
   190	}

   192	async function readPersistedLog(file: string): Promise<string> {
   193	  const content = await readFile(file)
   194	  if (!file.endsWith('.zstd')) return content.toString('utf8')
   195	  const scan = scanZstdFrames(content)
   196	  if (scan.tornStart !== undefined) throw new Error(`persisted snapshot log has a torn Zstandard frame: ${file}`)
   197	  const decoded: Buffer[] = []
   198	  for (const frame of scan.frames) {
   199	    decoded.push(await decompressZstdFrame(content.subarray(frame.start, frame.end)))
   200	  }
   201	  return Buffer.concat(decoded).toString('utf8')
   202	}

   204	async function persistedLogs(cwd: string, root: string = join(cwd, '.sessions')): Promise<PersistedLog[]> {
   205	  const files = (await readdir(root, { recursive: true }))
   206	    .filter(file => file.endsWith('.jsonl') || file.endsWith('.jsonl.zstd'))
   207	  return Promise.all(files.map(async (file) => {
   208	    const content = await readPersistedLog(join(root, file))
   209	    return { content, header: parseJsonl(content)[0] ?? {} }
   210	  }))
   211	}
